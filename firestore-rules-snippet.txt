// Merge these blocks into the existing Rules of the named database wadi-adai-db.
// Keep every existing rule used by the original service. Do not replace the full ruleset.
function isWadiAdaiEditor() {
  return request.auth != null
    && exists(/databases/$(database)/documents/wadi_adai_v2_members/$(request.auth.uid))
    && get(/databases/$(database)/documents/wadi_adai_v2_members/$(request.auth.uid)).data.active == true
    && get(/databases/$(database)/documents/wadi_adai_v2_members/$(request.auth.uid)).data.role == 'editor';
}
match /wadi_adai_v2/{section}/items/{recordId} {
  allow read: if true;
  allow create, update, delete: if isWadiAdaiEditor()
    && section in ['proposals', 'recommendations', 'stations', 'analyses', 'roadmap', 'observations', 'gallery'];
}
match /wadi_adai_v2_meta/{docId} {
  allow get: if true;
  allow list: if false;
  allow create: if isWadiAdaiEditor() && docId == 'bootstrap';
  allow update, delete: if false;
}
match /wadi_adai_v2_members/{uid} {
  allow get: if request.auth != null && request.auth.uid == uid;
  allow list, write: if false;
}
